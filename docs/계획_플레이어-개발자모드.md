# 플레이어/개발자 모드 분리 — 구현 계획

> **실행 방법:** 한 Task씩 진행. 각 Task는 ①코드 ②검증(헤드리스) ③F5 수동확인 ④커밋. 체크박스로 추적.
> **검증 도구(이 프로젝트 표준):**
> - 문법: `G=/Applications/Godot.app/Contents/MacOS/Godot; "$G" --headless --check-only --script <파일>` (Layout/Fx 등 오토로드 "Identifier not found"는 무시)
> - 씬 로드: `"$G" --headless res://scenes/<씬>.tscn --quit-after 40` → `script error`/`ERROR` 없으면 OK
> - 동작: 사용자가 F5로 확인

**Goal:** 순서대로 가는 플레이어 모드(저장/이어하기)와, 직업·등급·스테이지·난이도·치트를 자유 설정+도중 테스트하는 개발자 모드를 분리한다.

**Architecture:** `GameState`에 "런 설정(mode/difficulty/cheats/coins/level)" + 저장 시스템 + DEV 게이트를 둔다. 게임 본체(player/enemy/spawner)는 런 설정만 읽어 실행. 플레이어/개발자 모드는 런 설정을 채우는 입구(시작화면·개발자메뉴)만 다르다. 인게임 디버그 오버레이로 도중 테스트.

**Tech Stack:** Godot 4.6.3, GDScript. 설계 문서: `docs/설계_플레이어-개발자모드.md`.

## Task 1: GameState — DEV 게이트 + 런 설정 + Lv 배율 + 저장

**Files:** 수정 `scripts/game_state.gd`

- [ ] **Step 1: 런 설정/게이트/배율 필드 추가** (`var selected_job` 줄 위 또는 BUILD 아래에 삽입)

```gdscript
# --- 개발 게이트 ---  출시 빌드 만들 때 false 또는 OS.has_feature("dev")로 교체
const DEV := true
func is_dev() -> bool:
	return DEV

# --- 이번 판 런 설정 (게임 본체가 이것만 읽어 실행) ---
var mode: String = "player"        # "player" | "dev"
var difficulty: float = 1.0        # 적 스탯 배율 M (시스템밸런스 §3)
var cheats := {"godmode": false, "enemy_oneshot": false, "enemy_count_mult": 1.0}
var coins: int = 0                 # 재화(상점 시스템 때 사용)

# --- 등급(Lv) 배율 (시스템밸런스 §2.2) — hp/원거리/근거리에 곱함 ---
const LV_MULT := [1.0, 1.5, 2.2, 3.2, 4.5]
func level_mult(job: String = "") -> float:
	var j := job if job != "" else selected_job
	return LV_MULT[clampi(int(job_level.get(j, 1)) - 1, 0, 4)]
```

- [ ] **Step 2: 저장/불러오기 함수 추가** (파일 끝에)

```gdscript
const SAVE_PATH := "user://save.json"

func save_game() -> void:
	var data := {
		"stage_major": stage_major, "stage_minor": stage_minor,
		"jobs_unlocked": jobs_unlocked, "coins": coins,
	}
	var f := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if f:
		f.store_string(JSON.stringify(data))
		f.close()

func load_game() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var f := FileAccess.open(SAVE_PATH, FileAccess.READ)
	if f == null:
		return
	var data = JSON.parse_string(f.get_as_text())
	f.close()
	if typeof(data) == TYPE_DICTIONARY:
		stage_major = int(data.get("stage_major", 1))
		stage_minor = int(data.get("stage_minor", 1))
		jobs_unlocked = bool(data.get("jobs_unlocked", false))
		coins = int(data.get("coins", 0))

func reset_save() -> void:
	if FileAccess.file_exists(SAVE_PATH):
		DirAccess.remove_absolute(SAVE_PATH)
	reset_progress()
	coins = 0
```

- [ ] **Step 1: debug_overlay.gd 작성**

```gdscript
extends CanvasLayer
## DEV 전용 인게임 디버그 오버레이 — 🐞 버튼으로 패널 토글
var _panel: PanelContainer

func _ready() -> void:
	if not GameState.is_dev():
		queue_free()      # 출시 빌드: 아예 없음
		return
	layer = 100
	_build_ui()

func _build_ui() -> void:
	var btn := Button.new()
	btn.text = "🐞"
	btn.position = Vector2(8, 8)
	btn.pressed.connect(func(): _panel.visible = not _panel.visible)
	add_child(btn)
	_panel = PanelContainer.new()
	_panel.position = Vector2(8, 56)
	_panel.visible = false
	var box := VBoxContainer.new()
	_panel.add_child(box)
	add_child(_panel)
	_add(box, "코인 +1000", func(): GameState.coins += 1000)
	_add(box, "무적 토글", func(): GameState.cheats["godmode"] = not GameState.cheats.get("godmode", false))
	_add(box, "적 전멸", _kill_all)
	_add(box, "다음 스테이지", func():
		GameState.advance_stage()
		get_tree().reload_current_scene())
	# 상점/NPC/스킬은 시스템 생기면 여기에 버튼 추가

func _add(box: VBoxContainer, label: String, fn: Callable) -> void:
	var b := Button.new()
	b.text = label
	b.pressed.connect(fn)
	box.add_child(b)

func _kill_all() -> void:
	for e in get_tree().get_nodes_in_group("enemies"):
		if e.has_method("take_damage"):
			e.take_damage(999999.0)
```
